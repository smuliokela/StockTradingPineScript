f_adx(len) =>
    up = ta.change(high)
    down = -ta.change(low)
    plusDM = na(up) ? na : (up > down and up > 0 ? up : 0)
    minusDM = na(down) ? na : (down > up and down > 0 ? down : 0)
    trur = ta.rma(ta.tr, len)
    plus = fixnan(100 * ta.rma(plusDM, len) / trur)
    minus = fixnan(100 * ta.rma(minusDM, len) / trur)
    sum_dm = plus + minus
    adx_val = 100 * ta.rma(math.abs(plus - minus) / (sum_dm == 0 ? 1 : sum_dm), len)
    adx_val

f_vortex(len) =>
    vmp = math.sum(math.abs(high - low[1]), len)
    vmm = math.sum(math.abs(low - high[1]), len)
    str_val = math.sum(ta.tr(true), len)
    vip = vmp / str_val
    vim = vmm / str_val
    [vip, vim]

f_bbp() =>
    ema13 = ta.ema(close, 13)
    bull = high - ema13
    bear = low - ema13
    bull + bear

f_ao() =>
    ta.sma(hl2, 5) - ta.sma(hl2, 34)

f_uo() =>
    bp = low - math.min(low[1], close[1])
    tr = math.max(high, close[1]) - math.min(low, close[1])
    avg7 = math.sum(bp, 7) / math.sum(tr, 7)
    avg14 = math.sum(bp, 14) / math.sum(tr, 14)
    avg28 = math.sum(bp, 28) / math.sum(tr, 28)
    100 * (4 * avg7 + 2 * avg14 + avg28) / 7

f_smma(src, len) =>
    var float val = na
    val := na(val[1]) ? ta.sma(src, len) : (val[1] * (len - 1) + src) / len
    val

// Tori MTF logic is folded into f_evaluate_tf() below to eliminate duplicate MTF requests.
f_evaluate_tf() =>
    v_rsi = ta.rsi(close, rsi_len)
    rsi_is_rising = v_rsi > v_rsi[1]
    rsi_score = v_rsi > 70 ? -1 : v_rsi < 30 ? 1 : (v_rsi > 50 ? 1 : -1)

    stoch_raw = ta.stoch(close, high, low, stoch_k)
    stoch_k_val = ta.sma(stoch_raw, stoch_smooth)
    stoch_d_val = ta.sma(stoch_k_val, stoch_d)
    
    stoch_k_is_above_d_and_rising = (stoch_k_val > stoch_d_val) and (stoch_k_val > stoch_k_val[1])
    stoch_score = stoch_k_val > 80 ? -1 : stoch_k_val < 20 ? 1 : (stoch_k_val > 50 ? 1 : -1)

    [vip, vim] = f_vortex(vortex_len)
    vortex_plus_rising = vip > vip[1]
    vortex_minus_falling = vim < vim[1]
    vortex_bullish = (vip > vim) and vortex_plus_rising and vortex_minus_falling

    v_cci = ta.cci(close, cci_len)
    cci_score = v_cci > 100 ? 1 : v_cci < -100 ? -1 : (v_cci > 0 ? 1 : -1)

    v_adx = f_adx(adx_len)
    v_adx_prev = v_adx[1]
    adx_is_rising = v_adx > v_adx_prev
    adx_score = v_adx > 25 ? 1 : 0

    v_ao = f_ao()
    ao_score = v_ao > 0 ? 1 : -1

    v_mom = ta.mom(close, mom_len)
    mom_score = v_mom > 0 ? 1 : -1

    [macdV, macdSig, macdHist] = ta.macd(close, macd_fast, macd_slow, macd_sig)
    macd_score = macdV > macdSig ? 1 : -1

    rsiVal14 = ta.rsi(close, 14)
    stochRsi = ta.stoch(rsiVal14, rsiVal14, rsiVal14, 14)
    stochRsi_score = stochRsi > 80 ? -1 : stochRsi < 20 ? 1 : (stochRsi > 50 ? 1 : -1)

    v_wpr = ta.wpr(14)
    wpr_score = v_wpr > -20 ? 1 : v_wpr < -80 ? -1 : (v_wpr > -50 ? 1 : -1)

    v_bbp = f_bbp()
    bbp_score = v_bbp > 0 ? 1 : -1

    v_uo = f_uo()
    uo_score = v_uo > 50 ? 1 : -1

    osc_total = (rsi_score + stoch_score + cci_score + adx_score + ao_score + mom_score + macd_score + stochRsi_score + wpr_score + bbp_score + uo_score) / 11.0

    ema10  = ta.ema(close, 10)
    ema20  = ta.ema(close, 20)
    ema30  = ta.ema(close, 30)
    ema50  = ta.ema(close, 50)
    ema100 = ta.ema(close, 100)
    ema200 = ta.ema(close, 200)

    sma10  = ta.sma(close, 10)
    sma20  = ta.sma(close, 20)
    sma30  = ta.sma(close, 30)
    sma50  = ta.sma(close, 50)
    sma100 = ta.sma(close, 100)
    sma200 = ta.sma(close, 200)

    v_hma      = ta.hma(close, hma_len)
    v_vwma     = ta.vwma(close, vwma_len)
    v_ichimoku = math.avg(ta.highest(high, ichimoku_len), ta.lowest(low, ichimoku_len))

    e4   = ta.ema(close, 4)
    e10  = ta.ema(close, 10)
    e11  = ta.ema(close, 11)
    e20  = ta.ema(close, 20)
    e30  = ta.ema(close, 30)
    e50  = ta.ema(close, 50)
    e100 = ta.ema(close, 100)
    e200 = ta.ema(close, 200)

    ma_points = (e4 > e11 ? 10 : 0) + (e4 > e20 ? 10 : 0) + (e4 > e30 ? 10 : 0) + (e4 > e50 ? 10 : 0) +  (e4 > e100 ? 10 : 0) + (e4 > e200 ? 10 : 0) + (e4 > e4[1] ? 10 : 0) + (e11 > e20 ? 10 : 0) + (e11 > e30 ? 10 : 0) + (e11 > e50 ? 10 : 0) + (e11 > e100 ? 10 : 0) + (e11 > e200 ? 10 : 0) + (e20 > e30 ? 10 : 0) + (e20 > e50 ? 10 : 0) + (e20 > e100 ? 10 : 0) + (e20 > e200 ? 10 : 0) + (e30 > e50 ? 10 : 0) + (e30 > e100 ? 10 : 0) + (e30 > e200 ? 10 : 0) + (e50 > e100 ? 10 : 0) + (e50 > e200 ? 10 : 0) + (e50 > e50[1] ? 10 : 0) +  (e100 > e200 ? 10 : 0)
    
    ma_score_pct = (float(ma_points) / 230.0) * 100.0

    ma_score_sum = (close > e10  ? 1 : -1) + (close > e20  ? 1 : -1) + (close > e30  ? 1 : -1) +  (close > e50  ? 1 : -1) + (close > e100 ? 1 : -1) + (close > e200 ? 1 : -1) +  (close > sma10  ? 1 : -1) + (close > sma20  ? 1 : -1) + (close > sma30  ? 1 : -1) +  (close > sma50  ? 1 : -1) + (close > sma100 ? 1 : -1) + (close > sma200 ? 1 : -1) +  (close > v_hma  ? 1 : -1) + (close > v_vwma ? 1 : -1) + (close > v_ichimoku ? 1 : -1)

    ma_total = ma_score_sum / 15.0
    summary_score = (osc_total * osc_weight) + (ma_total * ma_weight)

    [diPlus, diMinus, adxValQuant] = ta.dmi(14, 14)
    sarVal = ta.sar(sarStart, sarInc, sarMax)
    
    jaw   = f_smma(hl2, jawLen)[8]
    teeth = f_smma(hl2, teethLen)[5]
    lips  = f_smma(hl2, lipsLen)[3]

    ts_adx       = (adxValQuant > 20 ? 10 : 0) + (adxValQuant > 25 ? 10 : 0) + (adxValQuant > adxValQuant[1] ? 10 : 0)
    ts_ma_stack  = (e4 > e11 ? 5 : 0) + (e11 > e20 ? 5 : 0) + (e20 > e50 ? 5 : 0) + (e50 > e200 ? 5 : 0)
    ts_ma_slope  = (e4 > e4[1] ? 5 : 0) + (e11 > e11[1] ? 5 : 0) + (e20 > e20[1] ? 5 : 0) + (e50 > e50[1] ? 5 : 0)
    ts_sar       = (close > sarVal ? 10 : 0)
    ts_alligator = ((lips > teeth) and (teeth > jaw) ? 10 : 0) + ((lips > lips[1]) and (teeth > teeth[1]) and (jaw > jaw[1]) ? 10 : 0)
    trend_score  = float(ts_adx + ts_ma_stack + ts_ma_slope + ts_sar + ts_alligator)

    mom_macd  = (macdHist > 0 ? 5 : 0) + (macdV > macdSig ? 5 : 0) + (macdV > macdV[1] ? 5 : 0) + (macdSig > macdSig[1] ? 5 : 0) + (math.abs(macdV - macdSig) > math.abs(macdV[1] - macdSig[1]) ? 5 : 0)
    mom_rsi   = (v_rsi > 50 ? 5 : 0) + (v_rsi > 60 ? 5 : 0) + (v_rsi > 70 ? 5 : 0) + (v_rsi > v_rsi[1] ? 5 : 0)
    
    stoch_diff      = math.abs(stoch_k_val - stoch_d_val)
    stoch_diff_prev = math.abs(stoch_k_val[1] - stoch_d_val[1])
    mom_stoch = (stoch_k_val > stoch_d_val ? 5 : 0) + (stoch_k_val > stoch_k_val[1] ? 10 : 0) + (stoch_d_val > stoch_d_val[1] ? 5 : 0) + (stoch_diff > stoch_diff_prev ? 5 : 0)
    
    mom_ind   = (v_mom > 0 ? 5 : 0) + (v_mom > v_mom[1] ? 5 : 0)
    mom_ao    = (v_ao > 0 ? 5 : 0) + (v_ao > v_ao[1] ? 5 : 0)
    mom_bonus = (macdV > macdSig) and (v_rsi > 50 and v_rsi > v_rsi[1]) and (stoch_k_val > stoch_d_val and stoch_k_val > stoch_k_val[1]) ? 10 : 0
    momentum_score = float(mom_macd + mom_rsi + mom_stoch + mom_ind + mom_ao + mom_bonus)

    bp_dmi      = (diPlus > diMinus ? 10 : 0) + (diPlus > diPlus[1] ? 5 : 0) + (diMinus < diMinus[1] ? 5 : 0)
    bp_vortex   = (vip > vim ? 10 : 0) + (vip > vip[1] ? 5 : 0) + (vim < vim[1] ? 5 : 0)
    bp_price    = (close > sarVal ? 5 : 0) + (close > e20 ? 5 : 0) + (close > e50 ? 5 : 0) + (close > e200 ? 5 : 0)
    bp_dir      = (macdV > macdSig ? 10 : 0) + (v_rsi > 50 ? 10 : 0) + (stoch_k_val > stoch_d_val ? 10 : 0)
    bp_bbp      = (v_bbp > 0 ? 10 : 0)
    bull_pressure_score = float(bp_dmi + bp_vortex + bp_price + bp_dir + bp_bbp)

    quant_score = (trend_score * w_trend) + (momentum_score * w_mom) + (bull_pressure_score * w_press)

    tf_e4   = ta.ema(close, len_ema4)
    tf_e11  = ta.ema(close, len_ema11)
    tf_e50  = ta.ema(close, len_ema50)
    tf_e200 = ta.ema(close, len_ema200)

    tf_up      = ta.change(high)
    tf_down    = -ta.change(low)
    tf_plusDM  = na(tf_up) ? na : (tf_up > tf_down and tf_up > 0 ? tf_up : 0)
    tf_minusDM = na(tf_down) ? na : (tf_down > tf_up and tf_down > 0 ? tf_down : 0)
    tf_tr      = ta.rma(ta.tr, di_len)
    tf_plus    = fixnan(100 * ta.rma(tf_plusDM, di_len) / tf_tr)
    tf_minus   = fixnan(100 * ta.rma(tf_minusDM, di_len) / tf_tr)
    tf_sum_di  = tf_plus + tf_minus
    tf_adx     = 100 * ta.rma(math.abs(tf_plus - tf_minus) / (tf_sum_di == 0 ? 1 : tf_sum_di), adx_len_3tf)

    tf_rsi = ta.rsi(close, rsi_len_3tf)
    [tf_macd, tf_macd_sig, _] = ta.macd(close, macd_fast_3tf, macd_slow_3tf, macd_sig_3tf)

    three_tf_bull =
         (tf_e4 > tf_e11) and
         (tf_e50 > tf_e200) and
         (tf_adx > adx_thresh) and
         (tf_plus > tf_minus) and
         (tf_rsi > rsi_thresh) and
         (tf_macd > tf_macd_sig) and
         (stoch_k_val > stoch_d_val)

    // ==========================================
    // ADX DIR ADDITION — preserves all original VCode13 logic
    // Bull: ADX > 25 + rising, DI+ rising, DI- falling,
    //       DI+ > DI-, RSI > 70
    // Bear: ADX > 25 + rising, DI- rising, DI+ falling,
    //       DI- > DI+, RSI < 30
    // ==========================================
    bool adx_dir_adx_rising = (v_adx > 25.0) and (v_adx > v_adx[1])
    bool adx_dir_bull = adx_dir_adx_rising and (diPlus > diPlus[1]) and (diMinus < diMinus[1]) and (diPlus > diMinus) and (v_rsi > 70.0)
    bool adx_dir_bear = adx_dir_adx_rising and (diMinus > diMinus[1]) and (diPlus < diPlus[1]) and (diMinus > diPlus) and (v_rsi < 30.0)

    // ----------------------------------------------------------
    // TORI MTF STATE — SAME LOGIC, PACKED INTO state_flags
    // This replaces a second request.security() pass on every TF.
    // Existing original behavior:
    // bias = HH+HL => +1, LH+LL => -1, otherwise 0
    // up_valid = last pivot low > previous pivot low
    // down_valid = last pivot high < previous pivot high
    // setup_state remains 0; touch_count remains 3
    // ----------------------------------------------------------
    float t_ph = ta.pivothigh(high, tori_piv_l, tori_piv_r)
    float t_pl = ta.pivotlow(low, tori_piv_l, tori_piv_r)
    float t_lastH = ta.valuewhen(not na(t_ph), t_ph, 0)
    float t_prevH = ta.valuewhen(not na(t_ph), t_ph, 1)
    float t_lastL = ta.valuewhen(not na(t_pl), t_pl, 0)
    float t_prevL = ta.valuewhen(not na(t_pl), t_pl, 1)

    int t_bias_val = 0
    if not (na(t_lastH) or na(t_prevH) or na(t_lastL) or na(t_prevL))
        t_bias_val := t_lastH > t_prevH and t_lastL > t_prevL ? 1 : t_lastH < t_prevH and t_lastL < t_prevL ? -1 : 0

    bool t_up_valid = t_lastL > t_prevL
    bool t_down_valid = t_lastH < t_prevH

    // Bit map:
    //   1 ADX rising
    //   2 Stoch bullish+rising
    //   4 Vortex bullish
    //   8 RSI rising
    //  16 3TF bullish
    //  32 Tori up-valid
    //  64 Tori down-valid
    // 128 Tori bias +1
    // 256 Tori bias -1
    int state_flags =
         (adx_is_rising ? 1 : 0) +
         (stoch_k_is_above_d_and_rising ? 2 : 0) +
         (vortex_bullish ? 4 : 0) +
         (rsi_is_rising ? 8 : 0) +
         (three_tf_bull ? 16 : 0) +
         (t_up_valid ? 32 : 0) +
         (t_down_valid ? 64 : 0) +
         (t_bias_val == 1 ? 128 : 0) +
         (t_bias_val == -1 ? 256 : 0) +
         (adx_dir_bull ? 512 : 0) +
         (adx_dir_bear ? 1024 : 0)

    [osc_total, ma_total, summary_score, v_adx, v_rsi, trend_score, momentum_score, bull_pressure_score, quant_score, ma_score_pct, state_flags]

f_get_rating(score) =>
    score >= th_strong_buy ? "STRONG BUY" : score >= th_buy ? "BUY" : score <= th_strong_sell ? "STRONG SELL" : score <= th_sell ? "SELL" : "NEUTRAL"

f_rating_color(r) =>
    r == "STRONG BUY" ? color.rgb(0, 200, 83) : r == "BUY" ? color.rgb(76, 175, 80) : r == "NEUTRAL" ? color.rgb(120, 120, 120) : r == "SELL" ? color.rgb(244, 67, 54) : r == "STRONG SELL" ? color.rgb(183, 28, 28) : color.gray

f_score_color(val) =>
    val >= 80.0 ? color.rgb(0, 230, 118) : val >= 65.0 ? color.rgb(76, 175, 80) : val >= 50.0 ? color.rgb(139, 195, 74) : val >= 40.0 ? color.rgb(255, 152, 0) : color.rgb(244, 67, 54)
